## Create a sandbox

An empty body uses the default image. REST returns `202` immediately by default; `?wait=ready` blocks until it's running and returns `201`.

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/sandbox?wait=ready" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"name": "web"}'
```

```shell CLI
bonya create web
```

```ts TypeScript
const sandbox = await client.createSandbox({ name: "web" });
```

```python Python
sandbox = client.create_sandbox(name="web")
```

```go Go
sandbox, err := client.Create(ctx, bonya.CreateOptions{Name: "web"})
```
</CodeGroup>

**MCP:** `sandbox_create`. Start from an image with `image`, from a snapshot with `snapshot`, or from a saved template with `template_id`.

## Get and list sandboxes

<CodeGroup>
```shell cURL
curl -s "https://api.bonya.ai/v1/sandbox?states=running&states=warm" \
  -H "Authorization: Bearer $BONYA_API_KEY"
```

```shell CLI
bonya list --state running,warm
bonya get web
```

```ts TypeScript
for await (const item of client.listSandboxes({ states: ["running", "warm"] })) {
  console.log(item.id, item.state);
}
const sandbox = await client.getSandboxByName("web");
```

```python Python
for item in client.list_sandboxes(states=["running", "warm"]):
    print(item.id, item.state)
sandbox = client.get_sandbox_by_name("web")
```

```go Go
for sb, err := range client.ListAll(ctx, bonya.ListOptions{States: []bonya.State{bonya.StateRunning, bonya.StateWarm}}) {
    if err != nil { log.Fatal(err) }
    fmt.Println(sb.ID, sb.State)
}
```
</CodeGroup>

**MCP:** `sandbox_list`, `sandbox_get`.

## Run a command (buffered exec)

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/exec" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"command": ["echo", "hello from bonya"]}'
```

```shell CLI
bonya exec web -- echo "hello from bonya"
```

```ts TypeScript
const result = await sandbox.exec(["echo", "hello from bonya"], { check: true });
console.log(result.stdout, result.exitCode);
```

```python Python
result = sandbox.exec(["echo", "hello from bonya"], check=True)
print(result.stdout, result.exit_code)
```

```go Go
result, err := sandbox.Exec(ctx, "echo", "hello from bonya")
fmt.Println(result.Stdout, result.ExitCode)
```
</CodeGroup>

**MCP:** `sandbox_exec`. A non-zero exit code is still a `200`.

## Stream a command (streaming exec)

<CodeGroup>
```ts TypeScript
import { Exit, Stderr, Stdout } from "@bonya-ai/bonya";

const session = sandbox.execStream(["bash", "-lc", "echo out; echo err >&2"]);
for await (const event of session) {
  if (event instanceof Stdout) process.stdout.write(Buffer.from(event.data));
  else if (event instanceof Stderr) process.stderr.write(Buffer.from(event.data));
  else if (event instanceof Exit) console.log("exit:", event.exitCode);
}
session.close();
```

```python Python
from bonya import Exit, Stderr, Stdout

with sandbox.exec_stream(["bash", "-lc", "echo out; echo err >&2"]) as session:
    for event in session:
        if isinstance(event, Stdout):
            print("out:", event.data.decode(), end="")
        elif isinstance(event, Stderr):
            print("err:", event.data.decode(), end="")
        elif isinstance(event, Exit):
            print("exit:", event.exit_code)
```

```go Go
stream, err := sandbox.ExecStream(ctx, bonya.ExecOptions{
    Command: []string{"bash", "-lc", "echo out; echo err >&2"},
    Stdout:  os.Stdout,
    Stderr:  os.Stderr,
})
status, err := stream.Wait()
fmt.Println("exit:", status.ExitCode)
```
</CodeGroup>

## Managed sessions (create, attach, list, kill)

A named, persistent command that outlives your connection, the counterpart to exec for dev servers and long builds.

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/sessions" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"name": "dev", "command": ["npm", "run", "dev"]}'
```

```shell CLI
bonya session create web dev -- npm run dev
bonya session attach web dev
bonya session list web
bonya session kill web dev
```

```ts TypeScript
await sandbox.createSession("dev", ["npm", "run", "dev"]);
const stream = await sandbox.attachSession("dev");
await stream.detach(); // the process keeps running
console.log(await sandbox.listSessions());
await sandbox.killSession("dev");
```

```python Python
sandbox.create_session("dev", ["npm", "run", "dev"])
with sandbox.attach_session("dev") as stream:
    ...  # leaving the block detaches
print(sandbox.list_sessions())
sandbox.kill_session("dev")
```

```go Go
sandbox.CreateSession(ctx, bonya.SessionOptions{Name: "dev", Command: []string{"npm", "run", "dev"}})
att, _ := sandbox.AttachSession(ctx, "dev", bonya.AttachOptions{Output: os.Stdout})
att.Detach()
sessions, _ := sandbox.ListSessions(ctx)
sandbox.KillSession(ctx, "dev", bonya.KillSessionOptions{})
```
</CodeGroup>

**MCP:** `session_create`, `session_list`, `session_kill`.

## Files

<CodeGroup>
```shell cURL
curl -s -X PUT "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/fs/write?path=/root/hello.txt" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/octet-stream" \
  --data-binary @./hello.txt

curl -s "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/fs/list?path=/root" \
  -H "Authorization: Bearer $BONYA_API_KEY"
```

```shell CLI
bonya fs upload web ./hello.txt /root/hello.txt
bonya fs ls web /root
bonya fs download web /root/hello.txt ./copy.txt
```

```ts TypeScript
await sandbox.writeFile("/root/hello.txt", "hello\n");
const data = await sandbox.readFile("/root/hello.txt");
const entries = await sandbox.listFiles("/root");
```

```python Python
sandbox.write_file("/root/hello.txt", "hello\n")
data = sandbox.read_file("/root/hello.txt")
entries = sandbox.list_files("/root")
```

```go Go
sandbox.WriteFile(ctx, "/root/hello.txt", []byte("hello\n"))
data, _ := sandbox.ReadFile(ctx, "/root/hello.txt", 0)
entries, _ := sandbox.ListDir(ctx, "/root")
```
</CodeGroup>

**MCP:** `fs_read`, `fs_write`, `fs_list`, `fs_stat`, `fs_mkdir`, `fs_rename`, `fs_delete`.

## Snapshots, forks, and lifecycle

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/snapshots" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"name": "baseline"}'

curl -s -X POST "https://api.bonya.ai/v1/sandbox/$SANDBOX_ID/fork" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"count": 2}'
```

```shell CLI
bonya snapshot create web --name baseline
bonya create restored --snapshot snp-...
```

```ts TypeScript
const snapshot = await sandbox.snapshot({ name: "baseline" });
const restored = await client.createSandbox({ snapshot: snapshot.id });
const { sandboxes } = await sandbox.fork(2);
await sandbox.pause("warm");
await sandbox.keepalive(1800);
```

```python Python
snapshot = sandbox.snapshot(name="baseline")
restored = client.create_sandbox(snapshot=snapshot.id)
forks = sandbox.fork(2).sandboxes
sandbox.pause("warm")
sandbox.keepalive(1800)
```

```go Go
snap, _ := sandbox.CreateSnapshot(ctx, bonya.SnapshotOptions{Name: "baseline"})
restored, _ := client.Create(ctx, bonya.CreateOptions{Snapshot: snap.ID})
fork, _ := sandbox.Fork(ctx, bonya.ForkOptions{Count: 2})
sandbox.Pause(ctx, bonya.PauseWarm)
sandbox.Keepalive(ctx, 30*time.Minute)
```
</CodeGroup>

**MCP:** `snapshot_*`, `sandbox_fork`, `sandbox_pause`, `sandbox_keepalive`.

## Jobs

A managed run of a command or script on a new or existing sandbox. `wait: true` blocks for the result; otherwise you get a run id to poll.

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/job" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"new_sandbox": {}, "command": ["./run-tests.sh"], "wait": true}'
```

```shell CLI
bonya job run --new --wait -- ./run-tests.sh
```

```ts TypeScript
const run = await client.runJob({ newSandbox: {}, cmd: ["./run-tests.sh"] });
console.log(run.status, run.result?.exitCode);
```

```python Python
from bonya import JobSandboxSpec, JobSpec

run = client.run_job(JobSpec(new_sandbox=JobSandboxSpec(), cmd=["./run-tests.sh"]))
print(run.status, run.result.exit_code)
```

```go Go
run, err := client.RunJob(ctx, bonya.JobSpec{
    NewSandbox: &bonya.JobNewSandbox{},
    Cmd:        []string{"./run-tests.sh"},
})
fmt.Println(run.Status, run.Result.ExitCode)
```
</CodeGroup>

## Job schedules

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/job-schedule" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"schedule": {"interval_seconds": 3600}, "job": {"new_sandbox": {}, "command": ["./nightly-report.sh"]}}'
```

```shell CLI
bonya schedule create --new --every 1h -- ./nightly-report.sh
bonya schedule trigger $SCHEDULE_ID
```

```ts TypeScript
const schedule = await client.createJobSchedule(
  { intervalSeconds: 3600 },
  { newSandbox: {}, cmd: ["./nightly-report.sh"] },
);
await client.triggerJobSchedule(schedule.scheduleId);
```

```python Python
from bonya import ScheduleSpec

schedule = client.create_job_schedule(
    ScheduleSpec(interval_seconds=3600),
    JobSpec(new_sandbox=JobSandboxSpec(), cmd=["./nightly-report.sh"]),
)
client.trigger_job_schedule(schedule.schedule_id)
```

```go Go
schedule, _ := client.CreateJobSchedule(ctx, bonya.JobScheduleSpec{
    IntervalSeconds: 3600,
    Job:             bonya.JobSpec{NewSandbox: &bonya.JobNewSandbox{}, Cmd: []string{"./nightly-report.sh"}},
})
client.TriggerJobSchedule(ctx, schedule.ID)
```
</CodeGroup>

## Templates, secrets, and registries

<CodeGroup>
```shell cURL
curl -s -X POST "https://api.bonya.ai/v1/templates" \
  -H "Authorization: Bearer $BONYA_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"name": "node-dev", "config": {"image": "node:22", "cpu": 2, "memory_mib": 4096}}'
```

```ts TypeScript
await client.createTemplate({ name: "node-dev", config: { image: "node:22", cpu: 2, memoryMib: 4096 } });
await client.createSecretSet({ name: "app-env", values: { API_URL: "https://example.com" } });
```

```python Python
client.create_template("node-dev", config=TemplateConfig(image="node:22", cpu=2, memory_mib=4096))
client.create_secret_set("app-env", {"API_URL": "https://example.com"})
```
</CodeGroup>

The CLI and the Go SDK only browse templates (`bonya template list`, `client.ListTemplates`). See [Templates](/concepts/templates) and [Secrets and registries](/concepts/secrets).
